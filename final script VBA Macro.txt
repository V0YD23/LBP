' Script to generate a 29x29 matrix of unit cells with RR values from Excel
' The cells will be spaced to avoid intersection

' Global parameters
Dim cellH, rectH, side, Z, H, spacing
Dim objExcel, objWorkbook, objWorksheet
Dim excelFile, row, col, RR, xpos, ypos
Dim outerRadiusTop, innerRadiusTop, halfRR, componentName

' Base parameters (used for all cells)
cellH = 0.018   ' Cell height
rectH = 0.254   ' Rectangle height
side = 3.5      ' Side length
Z = 0.272       ' Z offset
H = 3.0         ' Total height
spacing = 3.5   ' Spacing between cells (adjust as needed to prevent intersection)

' Excel file path - CHANGE THIS TO YOUR ACTUAL EXCEL FILE PATH
excelFile = "C:\Users\Yash\Desktop\LBP\phase_data_output.xlsx"

' Create Excel object
Set objExcel = CreateObject("Excel.Application")
objExcel.Visible = False  ' Keep Excel hidden

' Open the workbook
Set objWorkbook = objExcel.Workbooks.Open(excelFile)
Set objWorksheet = objWorkbook.Worksheets(1)  ' Use first worksheet

' Read RR values from Excel and create cells
For row = 1 To 29
    If (row = 14 Or row = 15 Or row = 16) Then
    For col = 1 To 29
        ' Read RR value from Excel
        RR = objWorksheet.Cells(row, col).Value

        ' Calculate position for this cell
        xpos = (col - 1) * spacing
        ypos = (row - 1) * spacing

        ' Calculate dimensions based on RR
        outerRadiusTop = (1.22 + 0.39 * RR) / 2
        innerRadiusTop = (2 - 0.45 * RR) / 2
        halfRR = RR / 2

        ' Create component name for this cell
        componentName = "cell_" & row & "_" & col

        ' Create new component for this cell
        StartVersionStringOverrideMode "2025.0|34.0.1|20240830"
        Component.New componentName

        ' Define cylinder: c1
        StartVersionStringOverrideMode "2025.0|34.0.1|20240830"
        With Cylinder
             .Reset
             .Name "c1"
             .Component componentName
             .Material "Copper (pure)"
             .OuterRadius CStr(outerRadiusTop)
             .InnerRadius CStr(innerRadiusTop)
             .Axis "z"
             .Zrange CStr(Z), CStr(cellH+Z)
             .Xcenter CStr(xpos)
             .Ycenter CStr(ypos)
             .Segments "0"
             .Create
        End With

        ' Define brick: rect1
        StartVersionStringOverrideMode "2025.0|34.0.1|20240830"
        With Brick
             .Reset
             .Name "rect1"
             .Component componentName
             .Material "Vacuum"
             .Xrange CStr(xpos-side/2), CStr(xpos+side/2)
             .Yrange CStr(ypos-side/2), CStr(ypos+side/2)
             .Zrange CStr(cellH+Z), CStr(cellH+rectH+Z)
             .Create
        End With

        ' Define brick: wing1
        StartVersionStringOverrideMode "2025.0|34.0.1|20240830"
        With Brick
             .Reset
             .Name "wing1"
             .Component componentName
             .Material "Copper (pure)"
             .Xrange CStr(xpos-outerRadiusTop), CStr(xpos-side/2)
             .Yrange CStr(ypos-0.2), CStr(ypos+0.2)
             .Zrange CStr(Z), CStr(cellH+Z)
             .Create
        End With

        ' Define brick: wing2
        StartVersionStringOverrideMode "2025.0|34.0.1|20240830"
        With Brick
             .Reset
             .Name "wing2"
             .Component componentName
             .Material "Copper (pure)"
             .Xrange CStr(xpos-0.2), CStr(xpos+0.2)
             .Yrange CStr(ypos-outerRadiusTop), CStr(ypos-side/2)
             .Zrange CStr(Z), CStr(cellH+Z)
             .Create
        End With

        ' Define brick: wing3
        StartVersionStringOverrideMode "2025.0|34.0.1|20240830"
        With Brick
             .Reset
             .Name "wing3"
             .Component componentName
             .Material "Copper (pure)"
             .Xrange CStr(xpos-0.2), CStr(xpos+0.2)
             .Yrange CStr(ypos+outerRadiusTop), CStr(ypos+side/2)
             .Zrange CStr(Z), CStr(cellH+Z)
             .Create
        End With

        ' Define brick: wing4
        StartVersionStringOverrideMode "2025.0|34.0.1|20240830"
        With Brick
             .Reset
             .Name "wing4"
             .Component componentName
             .Material "Copper (pure)"

        End With

        ' Define brick: rect2

             .Reset
             .Name "rect2"
             .Component componentName

             .Reset
             .Name "lower1"
             .Component componentName
             .Material "Copper (pure)"
